        }
    ];

    let currentProjectsCount = 3;  // initially show 3 projects
    const projectsGrid = document.getElementById('projectsGrid');
    const loadMoreBtn = document.getElementById('loadMoreBtn');

    function renderProjects(limit) {
        if (!projectsGrid) return;
        const toShow = projectsData.slice(0, limit);
        projectsGrid.innerHTML = '';
        toShow.forEach(project => {
            const card = document.createElement('div');
            card.className = 'project-card';
            // icon area
            const iconDiv = document.createElement('div');
            iconDiv.className = 'project-img';
            iconDiv.innerHTML = `<i class="${project.icon}" style="font-size: 3.2rem;"></i>`;
            // info area
            const infoDiv = document.createElement('div');
            infoDiv.className = 'project-info';
            const techSpans = project.tech.map(t => `<span class="tech-tag">${t}</span>`).join('');
            infoDiv.innerHTML = `
                <h3>${project.title}</h3>
                <p>${project.desc}</p>
                <div>${techSpans}</div>
                <a href="${project.demo}" class="project-link" target="_blank">Live demo <i class="fas fa-arrow-right"></i></a>
            `;
            card.appendChild(iconDiv);
            card.appendChild(infoDiv);
            projectsGrid.appendChild(card);
        });
    }

    // Load more or reset
    function handleLoadMore() {
        if (currentProjectsCount >= projectsData.length) {
            // if all are showing, optionally reset to 3? But better show all and toggle? Let's reset to initial count =3
            currentProjectsCount = 3;
            loadMoreBtn.innerHTML = '<i class="fas fa-expand-alt"></i> show all projects';
            renderProjects(currentProjectsCount);
        } else {
            currentProjectsCount = projectsData.length;
            renderProjects(currentProjectsCount);
            loadMoreBtn.innerHTML = '<i class="fas fa-compress-alt"></i> show less';
        }
    }

    // reset text if we click again from "show less" to initial
    function syncButtonText() {
        if (currentProjectsCount >= projectsData.length) {
            loadMoreBtn.innerHTML = '<i class="fas fa-compress-alt"></i> show less';
        } else {
            loadMoreBtn.innerHTML = '<i class="fas fa-expand-alt"></i> show all projects';
        }
    }

    if (loadMoreBtn) {
        loadMoreBtn.addEventListener('click', () => {
            if (currentProjectsCount >= projectsData.length) {
                currentProjectsCount = 3;
                renderProjects(currentProjectsCount);
                loadMoreBtn.innerHTML = '<i class="fas fa-expand-alt"></i> show all projects';
            } else {
                currentProjectsCount = projectsData.length;
                renderProjects(currentProjectsCount);
                loadMoreBtn.innerHTML = '<i class="fas fa-compress-alt"></i> show less';
            }
        });
    }

    // initial render (3 projects)
    renderProjects(3);

    // Dynamic typing effect for hero (extra JS spark)
    const createTypingEffect = () => {
        const heroParagraph = document.querySelector('.hero-content p');
        if (!heroParagraph) return;
        const originalText = heroParagraph.innerHTML;
        // We only want to replace the intro text? better add a dynamic span after name
        // but not to break existing layout. I will add a small inline span that changes roles.
        const nameStrong = heroParagraph.querySelector('strong');
        if (nameStrong) {
            const roleSpan = document.createElement('span');
            roleSpan.className = 'dynamic-role';
            roleSpan.style.display = 'inline-block';
            roleSpan.style.marginLeft = '6px';
            const roles = ['React Expert', 'UI Engineer', 'Node Developer', 'Creative Coder'];
            let idx = 0;
            function updateRole() {
                roleSpan.textContent = roles[idx % roles.length];
                idx++;
            }
            updateRole();
            setInterval(updateRole, 2200);
            // insert after strong
            nameStrong.insertAdjacentElement('afterend', document.createTextNode(' — '));
            nameStrong.insertAdjacentElement('afterend', roleSpan);
        }
    };

    // smooth anchor scroll
    document.querySelectorAll('.nav-links a, .hero-buttons a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function(e) {
            const href = this.getAttribute('href');
            if (href && href.startsWith('#')) {
                e.preventDefault();
                const targetId = href.substring(1);
                const targetElement = document.getElementById(targetId);
                if (targetElement) {
                    targetElement.scrollIntoView({ behavior: 'smooth', block: 'start' });
                }
            }
        });
    });

    // small hover effect on project demo link (just for fun)
    function addHoverDemo() {
        // not needed, but style included
    }

    // Initialize extra effect
    window.addEventListener('DOMContentLoaded', () => {
        createTypingEffect();
        // Add some style to make active link highlight
        const sections = document.querySelectorAll('section');
        const navLinks = document.querySelectorAll('.nav-links a');
        window.addEventListener('scroll', () => {
            let current = '';
            sections.forEach(section => {
                const sectionTop = section.offsetTop - 100;
                if (scrollY >= sectionTop) {
                    current = section.getAttribute('id');
                }
            });
            navLinks.forEach(link => {
                link.style.borderBottom = 'none';
                if (link.getAttribute('href') === `#${current}`) {
                    link.style.borderBottom = '2px solid #0f5c9e';
                    link.style.paddingBottom = '4px';
                } else {
                    link.style.borderBottom = 'none';
                }
            });
        });
    });
